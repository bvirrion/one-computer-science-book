\section*{Chapter \ref{ch:g2:using-a-computer} --- Using a Computer}

\begin{solution}{exo:g2:using-a-computer:1}
For example: the title bar, the button that closes it, the \omterm{def:g2:using-a-computer:menu}{menus}, the place
where the work is shown, the corner for resizing.
\end{solution}

\begin{solution}{exo:g2:using-a-computer:2}
\omterm{def:g2:keyboard-and-mouse:click}{Drag} its title bar to the new place. \omterm{def:g2:keyboard-and-mouse:click}{Drag} one of its corners outwards to make
it bigger (or \omterm{def:g2:keyboard-and-mouse:click}{click} on the button that makes it fill the screen).
\end{solution}

\begin{solution}{exo:g2:using-a-computer:3}
When we shut it down from the \omterm{def:g2:using-a-computer:menu}{menu}, the \omterm{def:g1:what-is-a-computer:computer}{computer} finishes writing what it was
keeping and closes everything properly. Pulling the plug can lose work that was
being written at that moment.
\end{solution}

\begin{solution}{exo:g2:using-a-computer:4}
Good: \code{volcano poster}, \code{class photo 2026}, \code{letter to grandma}.
Bad: \code{aaaa}, \code{work}, \code{xyz2} (they do not say what the work is).
\end{solution}

\begin{solution}{exo:g2:using-a-computer:5}
\menu{File > Save as}, with a new name such as \code{my story version 2}.
\menu{File > Save} would replace the first version.
\end{solution}

\begin{solution}{exo:g2:using-a-computer:6}
Save it, with \menu{File > Save as} and a good name such as
\code{castle drawing}, before closing the drawing and switching off --- and,
better, several times while he was drawing.
\end{solution}

\begin{solution}{exo:g2:using-a-computer:7}
The \omterm{def:g2:using-a-computer:window}{window} ``Drawing'' is in front. \omterm{def:g2:keyboard-and-mouse:click}{Click} on any visible part of the \omterm{def:g2:using-a-computer:window}{window}
``Story'' to bring it to the front.
\end{solution}

\begin{solution}{exo:g2:using-a-computer:8}
A \omterm{def:g2:using-a-computer:word-processor}{word processor}: it keeps sizes and bold letters. A \omterm{def:g2:using-a-computer:word-processor}{text editor} keeps only the
letters.
\end{solution}

\begin{solution}{exo:g2:using-a-computer:9}
For example, the names \code{story short version} and
\code{story long version}; or the names \code{story monday} and
\code{story wednesday}.
\end{solution}

\begin{solution}{exo:g2:using-a-computer:10}
First wait a little: the \omterm{def:g1:what-is-a-computer:computer}{computer} may be busy. If it still does not answer,
close the \omterm{def:g2:using-a-computer:window}{window} with its close button (an adult can help).
\end{solution}

\begin{solution}{exo:g2:using-a-computer:11}
Switch on the \omterm{def:g1:what-is-a-computer:computer}{computer}; wait until the \omterm{def:g2:using-a-computer:desktop}{desktop} appears; open the place where
you saved the drawing; \omterm{def:g2:keyboard-and-mouse:click}{double-click} on its name; go on drawing; save.
\end{solution}
