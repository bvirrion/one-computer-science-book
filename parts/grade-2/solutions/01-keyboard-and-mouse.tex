\section*{Chapter \ref{ch:g2:keyboard-and-mouse} --- The Keyboard and the Mouse}

\begin{solution}{exo:g2:keyboard-and-mouse:1}
(a)~\key{Shift}, held down while the letter \omterm{def:g2:keyboard-and-mouse:key}{key} is pressed (or \key{Caps Lock}
for many capitals). (b)~\key{Backspace}. (c)~\key{Enter}.
\end{solution}

\begin{solution}{exo:g2:keyboard-and-mouse:2}
It types the sign drawn above the 1 on that \omterm{def:g2:keyboard-and-mouse:key}{key}: on many keyboards an
exclamation mark ``!''. On some keyboards the digits themselves are the signs
typed with \key{Shift}, and \key{Shift} + 1 types the digit 1.
\end{solution}

\begin{solution}{exo:g2:keyboard-and-mouse:3}
The \omterm{def:g2:keyboard-and-mouse:cursor}{cursor} is in a text and shows where the next letter will appear; it moves
when you type or press the arrow \omterm{def:g2:keyboard-and-mouse:key}{keys}. The \omterm{def:g2:keyboard-and-mouse:pointer}{pointer} is the arrow that moves
with the mouse anywhere on the screen.
\end{solution}

\begin{solution}{exo:g2:keyboard-and-mouse:4}
(a)~A \omterm{def:g2:keyboard-and-mouse:click}{click}. (b)~A \omterm{def:g2:keyboard-and-mouse:click}{drag}. (c)~A \omterm{def:g2:keyboard-and-mouse:click}{double-click}.
\end{solution}

\begin{solution}{exo:g2:keyboard-and-mouse:5}
Sit with his back against the chair; put his feet flat on the floor or on a
footrest; move the screen (or himself) back to at least an arm's length.
\end{solution}

\begin{solution}{exo:g2:keyboard-and-mouse:6}
(a)~A tap. (b)~A swipe. (c)~A pinch, moving two fingers apart.
\end{solution}

\begin{solution}{exo:g2:keyboard-and-mouse:7}
\key{Backspace} at the top right, \key{Enter} at the right end of the middle
row, the space bar at the bottom in the middle. \key{Backspace} and \key{Enter}
type no letter (the space bar types a space).
\end{solution}

\begin{solution}{exo:g2:keyboard-and-mouse:8}
It replaces the mouse. Move one finger on it to move the \omterm{def:g2:keyboard-and-mouse:pointer}{pointer}; tap it (or
press its button) to \omterm{def:g2:keyboard-and-mouse:click}{click}.
\end{solution}

\begin{solution}{exo:g2:keyboard-and-mouse:9}
\key{Shift} held with H, then i, the space bar, \key{Shift} held with S, then
a, m.
\end{solution}

\begin{solution}{exo:g2:keyboard-and-mouse:10}
She should \omterm{def:g2:keyboard-and-mouse:click}{drag} it: press on the picture, move with the button held, release
at the new place. A \omterm{def:g2:keyboard-and-mouse:click}{click} only chooses the picture; it stays where it is.
\end{solution}

\begin{solution}{exo:g2:keyboard-and-mouse:11}
\key{Caps Lock} was on, so every letter came out as a capital, and the letters
typed with \key{Shift} came out small. He presses \key{Caps Lock} once to switch
it off, rubs the sentence out with \key{Backspace}, and types it again with
\key{Shift} only for the M and the L.
\end{solution}
