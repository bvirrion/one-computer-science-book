\section*{Chapter \ref{ch:g3:inside-the-computer} --- Inside the Computer}

\begin{solution}{exo:g3:inside-the-computer:1}
The \omterm{def:g3:inside-the-computer:processor}{processor}.
\end{solution}

\begin{solution}{exo:g3:inside-the-computer:2}
In the \omterm{def:g3:inside-the-computer:storage}{storage}: it keeps information when the power is off.
\end{solution}

\begin{solution}{exo:g3:inside-the-computer:3}
Everything he did since he last saved: it was only in the \omterm{def:g3:inside-the-computer:memory}{memory}, which forgets
when the power goes off.
\end{solution}

\begin{solution}{exo:g3:inside-the-computer:4}
The \omterm{def:g3:inside-the-computer:processor}{processor} (under its fan), the \omterm{def:g3:inside-the-computer:memory}{memory} sticks, the \omterm{def:g3:inside-the-computer:storage}{storage}, and, through
cables, the power supply; the ports at the back are part of the \omterm{def:g3:inside-the-computer:motherboard}{motherboard} too.
\end{solution}

\begin{solution}{exo:g3:inside-the-computer:5}
The fans push warm air out through the vents. If they are covered, the heat
stays inside: the laptop becomes very hot and slows down.
\end{solution}

\begin{solution}{exo:g3:inside-the-computer:6}
Printer: out. Microphone: in. Second screen: out. Headphones with a microphone:
both. Keyboard: in. Camera: in.
\end{solution}

\begin{solution}{exo:g3:inside-the-computer:7}
From the keyboard (an \omterm{def:g1:what-is-a-computer:device}{input device}) to the \omterm{def:g3:inside-the-computer:processor}{processor}, which works with the
\omterm{def:g3:inside-the-computer:memory}{memory}, then from the \omterm{def:g3:inside-the-computer:processor}{processor} to the screen (an \omterm{def:g1:what-is-a-computer:device}{output device}).
\end{solution}

\begin{solution}{exo:g3:inside-the-computer:8}
(a)~\omterm{def:g3:inside-the-computer:memory}{Memory}. (b)~\omterm{def:g3:inside-the-computer:storage}{Storage}. (c)~\omterm{def:g3:inside-the-computer:storage}{Storage}. (d)~\omterm{def:g3:inside-the-computer:memory}{Memory}. (e)~\omterm{def:g3:inside-the-computer:storage}{Storage}.
\end{solution}

\begin{solution}{exo:g3:inside-the-computer:9}
Through its case: the back of the tablet carries the heat out into the air,
which is why a tablet can feel warm in the hands. Its \omterm{def:g3:inside-the-computer:processor}{processor} is made to
produce little heat.
\end{solution}

\begin{solution}{exo:g3:inside-the-computer:10}
The battery replaces the power supply plugged into the wall: it gives the
electricity. A laptop needs one so that it can be carried and used away from a
plug.
\end{solution}
