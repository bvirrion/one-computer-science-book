\section*{Chapter \ref{ch:g5:text-pictures-sound-as-numbers} --- Text, Pictures and Sound as Numbers}

\begin{solution}{exo:g5:text-pictures-sound-as-numbers:1}
C is 67, A is 65, T is 84: 67, 65, 84.
\end{solution}

\begin{solution}{exo:g5:text-pictures-sound-as-numbers:2}
79 is O and 75 is K: ``OK''.
\end{solution}

\begin{solution}{exo:g5:text-pictures-sound-as-numbers:3}
They are different characters: the \omterm{def:g1:what-is-a-computer:computer}{computer} must know which one to show. A
\omterm{def:g5:text-pictures-sound-as-numbers:code}{code table} gives every character its own number.
\end{solution}

\begin{solution}{exo:g5:text-pictures-sound-as-numbers:4}
A black diamond in a white square:
\begin{center}
\begin{tikzpicture}[x=0.4cm, y=0.4cm]
  \foreach \row [count=\r from 0] in {00100,01110,11111,01110,00100}
    \foreach \c in {0,...,4} {
      \StrChar{\row}{\numexpr\c+1\relax}[\bitc]
      \if1\bitc \fill[black] (\c,-\r) rectangle ++(1,1); \fi
    }
  \draw[black!40] (0,-4) grid (5,1);
\end{tikzpicture}
\end{center}
\end{solution}

\begin{solution}{exo:g5:text-pictures-sound-as-numbers:7}
$10 \times 10 = 100$ \omterm{def:g5:text-pictures-sound-as-numbers:pixel}{pixels}, and $100 \times 3 = 300$ \omterm{def:g4:memory-and-storage:byte}{bytes}.
\end{solution}

\begin{solution}{exo:g5:text-pictures-sound-as-numbers:8}
10100101: black \omterm{def:g5:text-pictures-sound-as-numbers:pixel}{pixels} at the two edges (the outline of the face) and at the
third and sixth places: the two eyes.
\end{solution}

\begin{solution}{exo:g5:text-pictures-sound-as-numbers:9}
(255, 255, 0) is full red and full green light, no blue: yellow. (0, 255, 255)
is green and blue: cyan, a light blue-green.
\end{solution}

\begin{solution}{exo:g5:text-pictures-sound-as-numbers:10}
$100 \times 100 = \num{10000}$ \omterm{def:g5:text-pictures-sound-as-numbers:pixel}{pixels} against $200 \times 200 = \num{40000}$:
the second \omterm{def:g3:files-and-folders:file}{file} is 4 times bigger.
\end{solution}

\begin{solution}{exo:g5:text-pictures-sound-as-numbers:11}
The code of the character 7 is 55 (0 is 48, 1 is 49, \dots). The small numbers
of the table are already used for other things (the space is 32, and the
numbers below it are control codes), and a text must be able to hold the
character 7 without mixing it up with the number seven that a \omterm{def:g4:programs-and-operating-system:program}{program} counts
with.
\end{solution}
