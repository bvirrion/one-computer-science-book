\section*{Chapter \ref{ch:g4:memory-and-storage} --- Memory and Storage}

\begin{solution}{exo:g4:memory-and-storage:1}
The \omterm{def:g4:memory-and-storage:flash}{SSD}. The \omterm{def:g4:memory-and-storage:ram}{RAM} is emptied when the power goes off.
\end{solution}

\begin{solution}{exo:g4:memory-and-storage:2}
For example: a \omterm{def:g4:memory-and-storage:hard-disk}{hard disk} has moving parts (spinning disks and an arm), an \omterm{def:g4:memory-and-storage:flash}{SSD}
has none; an \omterm{def:g4:memory-and-storage:flash}{SSD} is faster; a \omterm{def:g4:memory-and-storage:hard-disk}{hard disk} hums, an \omterm{def:g4:memory-and-storage:flash}{SSD} is silent; a \omterm{def:g4:memory-and-storage:hard-disk}{hard disk}
can be scratched if it is knocked while running.
\end{solution}

\begin{solution}{exo:g4:memory-and-storage:3}
$3 \times \num{1000} = \num{3000}$ \omterm{def:g4:memory-and-storage:units}{kilobytes}.
\end{solution}

\begin{solution}{exo:g4:memory-and-storage:4}
1 \omterm{def:g4:memory-and-storage:units}{megabyte} is \num{1000} \omterm{def:g4:memory-and-storage:units}{kilobytes}, and $\num{1000} \div 2 = 500$ pages.
\end{solution}

\begin{solution}{exo:g4:memory-and-storage:5}
900 \omterm{def:g4:memory-and-storage:byte}{bytes}, 500 kB, 40 MB, 3 GB, 2 TB.
\end{solution}

\begin{solution}{exo:g4:memory-and-storage:6}
If the \omterm{def:g1:what-is-a-computer:computer}{computer} is lost, stolen, or broken by a fire or a flood, a copy kept
next to it may be lost at the same time. A copy kept elsewhere survives.
\end{solution}

\begin{solution}{exo:g4:memory-and-storage:7}
3 \omterm{def:g4:memory-and-storage:units}{megabytes}. 1 \omterm{def:g4:memory-and-storage:units}{gigabyte} is \num{1000} \omterm{def:g4:memory-and-storage:units}{megabytes}, and $\num{1000} \div 3$ is
333 with 1 left over: about 333 photos.
\end{solution}

\begin{solution}{exo:g4:memory-and-storage:8}
64 \omterm{def:g4:memory-and-storage:units}{gigabytes} are $64 \times \num{1000} = \num{64000}$ \omterm{def:g4:memory-and-storage:units}{megabytes}, and
$\num{64000} \div 4 = \num{16000}$ photos.
\end{solution}

\begin{solution}{exo:g4:memory-and-storage:9}
\num{12000} letters and spaces take \num{12000} \omterm{def:g4:memory-and-storage:byte}{bytes}, and
$\num{12000} \div \num{1000} = 12$ \omterm{def:g4:memory-and-storage:units}{kilobytes}.
\end{solution}

\begin{solution}{exo:g4:memory-and-storage:10}
The \omterm{def:g4:memory-and-storage:ram}{RAM} forgot it: her progress since the last save was only there, and the
\omterm{def:g4:memory-and-storage:ram}{RAM} is emptied when the power goes off. The progress she found again (the
first river) was the game's last save, kept in the \omterm{def:g3:inside-the-computer:storage}{storage}.
\end{solution}

\begin{solution}{exo:g4:memory-and-storage:11}
Many plans are good. For example: every Friday, the teacher copies the class
\omterm{def:g3:files-and-folders:folder}{folder} onto an external disk; there are two such disks, used in turn, and one
of them is always kept in another room or at home. A good plan says on what
the copy is made, where it is kept (not next to the \omterm{def:g1:what-is-a-computer:computer}{computer}), and how often
(at least once a week).
\end{solution}
