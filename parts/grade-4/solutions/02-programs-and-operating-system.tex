\begin{solution}{exo:g4:programs-and-operating-system:1}
A recipe card: no, its \omterm{def:g1:instructions-in-order:instruction}{instructions} are for a cook, not for a \omterm{def:g1:what-is-a-computer:computer}{computer}. A game:
yes. A photo: no, it is information, not \omterm{def:g1:instructions-in-order:instruction}{instructions}. A \omterm{def:g2:using-a-computer:word-processor}{text editor}: yes.
\end{solution}

\begin{solution}{exo:g4:programs-and-operating-system:2}

\omterm{def:g4:programs-and-operating-system:program}{programs}; the \omterm{def:g3:inside-the-computer:device}{devices}; the users and their accounts.
\end{solution}

\begin{solution}{exo:g4:programs-and-operating-system:5}
\omterm{def:g4:programs-and-operating-system:install}{Updates} fix mistakes that make \omterm{def:g4:programs-and-operating-system:program}{programs} stop or give wrong results, and they
close holes that strangers could use to get into the \omterm{def:g1:what-is-a-computer:computer}{computer}.
\end{solution}

\begin{solution}{exo:g4:programs-and-operating-system:6}
The user asks the \omterm{def:g4:programs-and-operating-system:application}{application} to print the page; the \omterm{def:g4:programs-and-operating-system:application}{application} asks the
\omterm{def:g4:programs-and-operating-system:os}{operating system} to send the page to the printer; the \omterm{def:g4:programs-and-operating-system:os}{operating system} drives
the printer (a \omterm{def:g3:inside-the-computer:device}{device}). The answer ``printed'' comes back up the same way.
\end{solution}

(e)~\omterm{def:g4:programs-and-operating-system:os}{Operating system}.

\begin{solution}{exo:g4:programs-and-operating-system:8}
Each one's \omterm{def:g3:files-and-folders:file}{files} and \omterm{def:g3:files-and-folders:folder}{folders}, settings and password. Each sees only their own
work, cannot delete the other's by mistake, and keeps their own things private.
\end{solution}

\begin{solution}{exo:g4:programs-and-operating-system:9}
The \omterm{def:g3:inside-the-computer:processor}{processor} reads its \omterm{def:g1:instructions-in-order:instruction}{instructions} from the \omterm{def:g3:inside-the-computer:memory}{memory}, which is much faster
than the \omterm{def:g3:inside-the-computer:storage}{storage}; the \omterm{def:g3:inside-the-computer:storage}{storage} only keeps the \omterm{def:g4:programs-and-operating-system:program}{program} while it is not running.
\end{solution}

\begin{solution}{exo:g4:programs-and-operating-system:10}
Close the window without downloading anything, and tell an adult. Real \omterm{def:g4:programs-and-operating-system:install}{updates}
come from the \omterm{def:g4:programs-and-operating-system:os}{operating system} or from the places the \omterm{def:g3:inside-the-computer:device}{device} trusts, not from
a window that appears on a website; such a window may try to \omterm{def:g4:programs-and-operating-system:install}{install} a harmful
\omterm{def:g4:programs-and-operating-system:program}{program}.
\end{solution}
