\section*{Chapter \ref{ch:g1:instructions-in-order} --- Instructions in Order}

\begin{solution}{exo:g1:instructions-in-order:1}
Underpants, trousers, socks, shoes, coat. (Socks may also come before the trousers;
the shoes must come after the socks and after the trousers.)
\end{solution}

\begin{solution}{exo:g1:instructions-in-order:2}
Can be swapped: the trousers and the socks. Cannot be swapped: the socks and
the shoes (shoes first, and the socks cannot go on).
\end{solution}

\begin{solution}{exo:g1:instructions-in-order:3}
Two squares to the right, a turn to face up, three squares up: the robot stops
on C.
\end{solution}

\begin{solution}{exo:g1:instructions-in-order:4}
For example: forward five times, left, forward four times:
$\uparrow\uparrow\uparrow\uparrow\uparrow\ \circlearrowleft\ \uparrow\uparrow\uparrow\uparrow$
(10 \omterm{def:g1:instructions-in-order:instruction}{instructions}).
\end{solution}

\begin{solution}{exo:g1:instructions-in-order:5}
A is three squares to the right of S, so the fourth \omterm{def:g1:instructions-in-order:instruction}{instruction} must be
``left'', not ``forward''. One ``forward'' too many at the start:
$\uparrow\ \uparrow\ \uparrow\ \circlearrowleft\ \uparrow\ \uparrow$.
\end{solution}

\begin{solution}{exo:g1:instructions-in-order:6}
(b). Sequence (a) turns right after two squares and leaves the grid at the
bottom. Sequence (c) also ends on the star, but by another path: it goes up
from the second column, not from the third.
\end{solution}

\begin{solution}{exo:g1:instructions-in-order:7}
Both end on A. The first has 6 \omterm{def:g1:instructions-in-order:instruction}{instructions}, the second 8.
\end{solution}

\begin{solution}{exo:g1:instructions-in-order:8}
The walls block the two left squares of the middle row, so the robot must go
up in the third column or farther right:
$\uparrow\uparrow\ \circlearrowleft\ \uparrow\uparrow\uparrow\uparrow\ \circlearrowleft\ \uparrow\uparrow$,
10 \omterm{def:g1:instructions-in-order:instruction}{instructions}. It needs at least 2 squares to the right, 4 up and 2 back to
the left (8 forwards) and at least 2 turns, so 10 is the shortest.
\end{solution}

\begin{solution}{exo:g1:instructions-in-order:9}
For example $\uparrow\ \uparrow\ \circlearrowright\ \circlearrowright\ \uparrow\ \uparrow$
(two squares out, a half turn, two squares back), or
$\uparrow\ \circlearrowleft\ \uparrow\ \circlearrowleft\ \uparrow\ \circlearrowleft\ \uparrow$
(around a small square).
\end{solution}

\begin{solution}{exo:g1:instructions-in-order:10}
Three right turns make the same turn as one left turn:
$\uparrow\uparrow\uparrow\ \circlearrowright\circlearrowright\circlearrowright\ \uparrow\uparrow$.
\end{solution}
